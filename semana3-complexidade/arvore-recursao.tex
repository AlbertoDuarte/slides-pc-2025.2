        \[ \lim_{n \to \infty} \frac{f(n)}{g(n)} = c \ . \]
        Então
            \begin{itemize}
                \item[(i)] se $c > 0$, então $f(n) = \Theta(g(n))$;
                \item[(ii)] se $c = 0$, então $f(n) = O(g(n))$; % o(g(n)) na vdd
                \item[(iii)] se $c = \infty$, então $g(n) = O(g(n))$;  
            \end{itemize}
    \end{block}
