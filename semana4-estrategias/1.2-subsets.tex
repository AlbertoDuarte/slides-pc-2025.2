\begin{frame}{Subconjuntos}
    \metroset{block=fill}

    \begin{block}{Problema}
        Dado um conjunto de $n \leq 20$ inteiros, calcule a maior soma de um subconjunto cuja soma seja $\leq X$ para um inteiro $X$ dado.
    \end{block} \pause
    
    {\bf Exemplo.} O conjunto $\{5, 7, 11\}$ tem $2^3 = 8$ subconjuntos: \pause
    \begin{itemize}
        \item $\{\}$, \pause
        \item $\{5\}$, $\{7\}$, $\{11\}$, \pause
        \item $\{5, 7\}$, $\{5, 11\}$, $\{7, 11\}$, \pause
        \item $\{5, 7, 11\}$. \pause
    \end{itemize}
    Se $X = 13$, a resposta é $12$. \pause 

\end{frame}
