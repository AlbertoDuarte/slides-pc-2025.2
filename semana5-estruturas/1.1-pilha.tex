\begin{frame}{Pilha}
    \metroset{block=fill}

    \begin{block}{Definição}
        Uma {\bf pilha} é uma estrutura onde a remoção remove o elemento mais recentemente inserido.
    \end{block}

    \begin{itemize}
        \item Acesso: $O(1)$.
        \item Inserção: $O(1)$.
        \item Remoção: $O(1)$.
    \end{itemize}
